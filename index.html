PostgreSQL Relational Schema & Row Level Security (RLS)

Production Supabase DDL blueprint powering ScholarTrack's role-based data isolation

1. Normalized 3NF Tables

17 relational tables separating profiles, students, faculty, departments, enrollments, attendance_sessions, and marks.

2. Database-Level RBAC

Supabase RLS policies ensure students can only SELECT their own rows and cannot mutate attendance or marks.

3. Single Source of Truth

Updates committed in the Faculty or Admin portals immediately reflect in the Student dashboard calculations.

-- ScholarTrack Relational Schema & Row Level Security (Supabase PostgreSQL)
CREATE TABLE profiles (
  id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  full_name TEXT NOT NULL,
  email TEXT UNIQUE NOT NULL,
  role TEXT NOT NULL CHECK (role IN ('student', 'faculty', 'admin')),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE departments (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name TEXT NOT NULL,
  code TEXT UNIQUE NOT NULL
);

CREATE TABLE students (
  id UUID PRIMARY KEY REFERENCES profiles(id) ON DELETE CASCADE,
  roll_number TEXT UNIQUE NOT NULL,
  department_id UUID REFERENCES departments(id),
  semester INT NOT NULL CHECK (semester BETWEEN 1 AND 8),
  section TEXT NOT NULL DEFAULT 'A',
  is_active BOOLEAN DEFAULT TRUE
);

CREATE TABLE subjects (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  code TEXT UNIQUE NOT NULL,
  name TEXT NOT NULL,
  credits INT NOT NULL DEFAULT 4,
  department_id UUID REFERENCES departments(id),
  semester INT NOT NULL
);

CREATE TABLE attendance_sessions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  subject_id UUID REFERENCES subjects(id) ON DELETE CASCADE,
  faculty_id UUID REFERENCES profiles(id),
  session_date DATE NOT NULL,
  topic TEXT
);

CREATE TABLE attendance_records (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  session_id UUID REFERENCES attendance_sessions(id) ON DELETE CASCADE,
  student_id UUID REFERENCES students(id) ON DELETE CASCADE,
  status TEXT NOT NULL CHECK (status IN ('present', 'absent')),
  UNIQUE(session_id, student_id)
);

CREATE TABLE marks (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  subject_id UUID REFERENCES subjects(id) ON DELETE CASCADE,
  student_id UUID REFERENCES students(id) ON DELETE CASCADE,
  assessment_type TEXT NOT NULL,
  marks_obtained NUMERIC(5,2) NOT NULL,
  max_marks NUMERIC(5,2) NOT NULL,
  is_published BOOLEAN DEFAULT FALSE
);

-- Enable Row Level Security (RLS)
ALTER TABLE attendance_records ENABLE ROW LEVEL SECURITY;
ALTER TABLE marks ENABLE ROW LEVEL SECURITY;

-- Policy: Students can ONLY read their own attendance records
CREATE POLICY "student_read_own_attendance" ON attendance_records
  FOR SELECT USING (auth.uid() = student_id);

-- Policy: Students can ONLY read their own PUBLISHED marks
CREATE POLICY "student_read_published_marks" ON marks
  FOR SELECT USING (auth.uid() = student_id AND is_published = TRUE);

-- Policy: Faculty can manage attendance & marks for assigned subjects
CREATE POLICY "faculty_manage_subject_records" ON attendance_records
  FOR ALL USING (
    EXISTS (SELECT 1 FROM profiles WHERE id = auth.uid() AND role IN ('faculty', 'admin'))
  );
